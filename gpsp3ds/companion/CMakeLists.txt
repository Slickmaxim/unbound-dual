cmake_minimum_required(VERSION 3.18)
project(gpsp_companion C)
set(CMAKE_C_STANDARD 11)
file(GLOB VENDOR "vendor/*.c")
add_library(companion STATIC ${VENDOR} alloc.c companion.c readonly_ui.c adapter.c cache.c progress.c localization.c)
set_source_files_properties(${VENDOR} PROPERTIES COMPILE_DEFINITIONS "CP_REDIRECT_ALLOC")
target_include_directories(companion PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/vendor")
target_compile_definitions(companion PRIVATE PD_UNBOUND_ONLY)
target_compile_options(companion PRIVATE -O2 -ffunction-sections -fdata-sections -include "${CMAKE_CURRENT_SOURCE_DIR}/alloc.h")
if(COMPANION_NATIVE)
 target_compile_definitions(companion PRIVATE COMPANION_NATIVE)
endif()
if(CMAKE_SYSTEM_PROCESSOR STREQUAL "arm")
 target_compile_definitions(companion PRIVATE __3DS__)
endif()
if(COMPANION_TESTS)
 add_executable(companion-fixture-test test/fixture_test.c ../source/ui.c ../source/ui_font.c)
 target_include_directories(companion-fixture-test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/vendor")
 target_compile_definitions(companion-fixture-test PRIVATE PD_UNBOUND_ONLY CP_ENABLED)
 if(COMPANION_NATIVE)
  target_compile_definitions(companion-fixture-test PRIVATE COMPANION_NATIVE)
 endif()
 target_compile_options(companion-fixture-test PRIVATE -UNDEBUG -ffunction-sections -fdata-sections)
 target_link_options(companion-fixture-test PRIVATE -Wl,--gc-sections)
 target_link_libraries(companion-fixture-test companion m)
endif()
